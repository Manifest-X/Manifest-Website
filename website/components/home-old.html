import { useState, FormEvent } from 'react'
  
                  export default function Tasks() {
                  const [tasks, setTasks] = useState<string[]>([
                  'Try Manifest', 'Read the docs'
                  ])
                  const [input, setInput] = useState('')
  
                  function submit(e: FormEvent) {
                  e.preventDefault()
                  if (input) {
                  setTasks([...tasks, input])
                  setInput('')
                  }
                  }
  
                  return (
                  <>
                  <ul>
                  {tasks.map((t, i) => (
                  <li key={i}>
                  <span>{t}</span>
                  <button onClick={() =>
                  setTasks(tasks.filter((_, j) => j !== i))
                  }>×</button>
                  </li>
                  ))}
                  </ul>
                  <form onSubmit={submit}>
                  <input value={input}
                  onChange={e => setInput(e.target.value)}/>
                  <button type="submit">Add</button>
                  </form>
                  </>
                  )
                  }

Capabilities

Everything a site needs. Nothing it doesn't.

Components

<x-name> resolves from your components/ folder. No imports.

Reactive data

$x.foo reads YAML, JSON, CSV, or any API. Search and query magic included.

Static rendering

npx mnfst-render bakes every route to HTML with meta, OG, and llms.txt.

Localization

One source, many locales. URL-routed, lang/dir handled, hreflang generated.

Markdown

<x-markdown> renders .md files with syntax highlighting and frontmatter.

Theme + Tailwind

CSS custom-property theme. Drop in Tailwind utilities. Both work side-by-side.
